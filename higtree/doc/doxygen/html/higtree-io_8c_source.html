96hig_cell * higio_read_from_amr_info(higio_amr_info *mi)
97{
98 hig_cell *root = hig_create_root(mi->l, mi->h);
99
100 for(int i = 0; i < DIM; i++) {
101 assert(FLT_EQ(mi->levels[0].delta[i],
102 (mi->h[i] - mi->l[i])/mi->levels[0].patches[0].patchsize[i]));
103 }
104
105 hig_refine_uniform(root, mi->levels[0].patches[0].patchsize);
106
107 int nc[DIM];
108 POINT_ASSIGN_SCALAR(nc, 2);
109 for(int l = 1; l < mi->numlevels; l++) {
110 __higio_level_info *level = mi->levels + l;
111 for(int p = 0; p < level->numpatches; p++) {
112 __higio_patch_info *patch = level->patches + p;
113
114 Point initpatch;
115 POINT_MULT(initpatch, level->delta, patch->initcell);
116 POINT_ADD(initpatch, initpatch, mi->l);
117
118 int k[DIM];
119 POINT_ASSIGN_SCALAR(k, 0);
120 while(1) {
121 Point offset;
122 POINT_MULT(offset, level->delta, k);
123 Point point;
124 POINT_ADD(point, initpatch, offset);
125 hig_cell *c = hig_get_cell_with_point(root, point);
126 Point cdelta;
127 hig_get_delta(c, cdelta);
128 int isrefinedenough = 1;
129 for(int i = 0; i < DIM; i++) {
130 if(FLT_LT(level->delta[i], cdelta[i])) {
131 isrefinedenough = 0;
132 break;
133 }
134 }
135 if(!isrefinedenough) {
136 hig_refine_uniform(c, nc);
137 }
138 int i = 0;
139 while(i < DIM && k[i] + 2 >= patch->patchsize[i] - 1) {
140 k[i] = 0;
141 i++;
142 }
143 if (i < DIM) {
144 k[i]+=2;
145 } else {
146 break;
147 }
148 }
149 }
150 }
151 return root;
152}

336void higio_compute_cells_per_coarser_block_from_amr_info(higio_amr_info *mi, int numcells[], int numblocks) {
337 for(int c = 0; c < numblocks; c++) {
338 numcells[c] = 1;
339 }
340 int div = 1;
341 int nc[DIM];
342 POINT_ASSIGN(nc, mi->levels[0].patches[0].patchsize);
343 for(int l = 1; l < mi->numlevels; l++) {
344 for(int p = 0; p < mi->levels[l].numpatches; p++) {
345 assert(mi->levels[l].patches[p].initcell[0] % 2 == 1);
346 int cl[DIM];
347 int ch[DIM];
348 for(int d = 0; d < DIM; d++) {
349 cl[d] = mi->levels[l].patches[p].initcell[d] - 1;
350 ch[d] = cl[d] + mi->levels[l].patches[p].patchsize[d];
351 cl[d] >>= 1;
352 ch[d] >>= 1;
353 }
354 int ind[DIM];
355 POINT_ASSIGN(ind, cl);
356 while(1) {
357 int pos = 0;
358 for(int dim = 0; dim < DIM; dim++) {
359 pos *= nc[dim];
360 pos += (ind[dim] >> (l-1));
361 }
362 numcells[pos] += (1<<DIM)-1;
363 int dim = 0;
364 while(dim < DIM && ind[dim] >= ch[dim] - 1) {
365 ind[dim] = cl[dim];
366 dim++;
367 }
368 if (dim < DIM) {
369 ind[dim]++;
370 } else {
371 break;
372 }
373 }
374 }
375 }

393hig_cell *higio_read_bc_from_amr(higio_amr_info *mi, int bcdim, int bcdir) {
394 Point lowpoint, highpoint;
395 POINT_ASSIGN(lowpoint, mi->l);
396 POINT_ASSIGN(highpoint, mi->h);
397 if (bcdir == 0) {
398 lowpoint[bcdim] = mi->l[bcdim] - EPSDELTA;
399 highpoint[bcdim] = mi->l[bcdim] + EPSDELTA;
400 } else {
401 lowpoint[bcdim] = mi->h[bcdim] - EPSDELTA;
402 highpoint[bcdim] = mi->h[bcdim] + EPSDELTA;
403 }
404 hig_cell *root = hig_create_root(lowpoint, highpoint);
405 int nc[DIM];
406 POINT_ASSIGN(nc, mi->levels[0].patches[0].patchsize);
407 nc[bcdim] = 1;
408 hig_refine_uniform(root, nc);
409 POINT_ASSIGN_SCALAR(nc, 2);
410 nc[bcdim] = 1;
411 for(int l = 1; l < mi->numlevels; l++) {
412 Point delta;
413 POINT_ASSIGN(delta, mi->levels[l].delta);
414 int numpatches = mi->levels[l].numpatches;
415 for(int p = 0; p < numpatches; p++) {
416 int initcell[DIM];
417 int patchsize[DIM];
418 POINT_ASSIGN(initcell, mi->levels[l].patches[p].initcell);
419 int istorefine = 0;
420 if(bcdir == 0) {
421 if(initcell[bcdim] == 1) {
422 istorefine = 1;
423 }
424 } else {
425 int lastcell = initcell[bcdim] + mi->levels[l].patches[p].patchsize[bcdim];
426 //DEBUG_INSPECT(lastcell, %d);
427 //DEBUG_INSPECT(mi->levels[0].patches[0].patchsize[bcdim] * (1 << l) + 1, %d);
428 if (lastcell == mi->levels[0].patches[0].patchsize[bcdim] * (1 << l) + 1) {
429 istorefine = 1;
430 }
431 }
432 if (istorefine) {
433 POINT_ASSIGN(patchsize, mi->levels[l].patches[p].patchsize);
434 patchsize[bcdim] = 1;
435 Point initpatch;
436 POINT_MULT(initpatch, delta, initcell);
437 POINT_ADD(initpatch, initpatch, mi->l);
438 int k[DIM];
439 int firstid[DIM];
440 POINT_ASSIGN_SCALAR(firstid, 0);
441 int lastid[DIM];
442 POINT_ASSIGN(lastid, patchsize);
443 POINT_SUB_SCALAR(lastid, lastid, 1);
444 if (bcdir == 0) {
445 lastid[bcdim] = firstid[bcdim];
446 } else {
447 firstid[bcdim] = lastid[bcdim];
448 }
449 POINT_ASSIGN(k, firstid);
450 while(1) {
451 Point offset;
452 POINT_MULT(offset, delta, k);
453 Point point;
454 POINT_ADD(point, initpatch, offset);
455 if(bcdir == 0) {
456 point[bcdim] = lowpoint[bcdim];
457 } else {
458 point[bcdim] = highpoint[bcdim];
459 }
460 hig_cell *c = hig_get_cell_with_point(root, point);
461 if (c != NULL) {
462 Point cdelta;
463 hig_get_delta(c, cdelta);
464 int isrefinedenough = 1;
465 for(int dim = 0; dim < DIM; dim++) {
466 if(FLT_LT(delta[dim], cdelta[dim])) {
467 isrefinedenough = 0;
468 break;
469 }
470 }
471 if(!isrefinedenough) {
472 hig_refine_uniform(c, nc);
473 }
474 }
475 int dim = 0;
476 while(dim < DIM && k[dim] + 2 >= lastid[dim]) {
477 k[dim] = firstid[dim];
478 dim++;
479 }
480 if (dim < DIM) {
481 k[dim]+=2;
482 } else {
483 break;
484 }
485 }
486 }
487 }
488 }
489 return root;
490}

597 dt_higcell = H5Tcreate(H5T_COMPOUND, sizeof(hig_serial_tree));
598 H5Tinsert(dt_higcell, "lowpoint",
599 HOFFSET(hig_serial_tree, lowpoint), dt_point);
600 H5Tinsert(dt_higcell, "highpoint",
601 HOFFSET(hig_serial_tree, highpoint), dt_point);
602 H5Tinsert(dt_higcell, "numcells",
603 HOFFSET(hig_serial_tree, numcells), dt_cellcount);
604 }
605
606 H5Tclose(dt_cellcount);
607 }
608
609 // The dataspace (memory accessor and the on-file layout)
610 hid_t ds_mem, ds_file;
611 {
612 hsize_t unlimited = H5S_UNLIMITED;
613 ds_mem = H5Screate_simple(1, &chunk_size, &unlimited);
614 ds_file = H5Scopy(ds_mem);
615 }
616
617 // Prepare the dataset to receive data
618 hid_t dset;
619 {
620 hid_t dset_plist = H5Pcreate(H5P_DATASET_CREATE);
621 H5Pset_chunk(dset_plist, 1, &chunk_size);
622
623 unsigned int compression_level = 1; // Fastest compression
624 H5Pset_filter(dset_plist, H5Z_FILTER_DEFLATE, H5Z_FLAG_OPTIONAL, 1, &compression_level);
625
626 /* Create the dataspace and the chunked dataset */
627 dset = H5Dcreate(ctx->fd, path, dt_higcell, ds_mem, H5P_DEFAULT, dset_plist, H5P_DEFAULT);
628 H5Pclose(dset_plist);
629 }
630
631 // Model the data to the correct format
632 {
633 hig_serial_tree buff[chunk_size];
634
635 higcit_celliterator *iter;
636 hsize_t i = 0;
637 hsize_t written_size = 0;
638
639 for (iter = higcit_create_breadth_first(root); !higcit_isfinished(iter); higcit_nextcell(iter))
640 {
641 hig_cell *c = higcit_getcell(iter);
642 hig_get_lowpoint(c, buff[i].lowpoint);
643 hig_get_highpoint(c, buff[i].highpoint);
644 memcpy(buff[i].numcells, c->numcells, sizeof(buff[i].numcells));
645
646 hig_cell *parent = hig_get_parent(c);
647
648 if(++i == chunk_size) {
649 hsize_t offset = written_size;
650 i = 0;
651
652 // Set the new size of the dataset
653 written_size += chunk_size;
654 H5Dset_extent(dset, &written_size);
655
656 // Configure the hyperslab of where to write
657 H5Sclose(ds_file);
658 ds_file = H5Dget_space(dset);
659 H5Sselect_hyperslab(ds_file, H5S_SELECT_SET, &offset, NULL, &chunk_size, NULL);
660
661 H5Dwrite(dset, dt_higcell, ds_mem, ds_file, H5P_DEFAULT, buff);
662 }
663 }
664
665 // Last write, incomplente chunk...
666 if(i) {
667 hsize_t offset = written_size;
668
669 // Reconfigure memory dataspace to fit the number of elements in buffer
670 hsize_t unlimited = H5S_UNLIMITED;
671 H5Sset_extent_simple(ds_mem, 1, &i, &unlimited);
672
673 // Set the new size of the dataset
674 written_size += i;
675 H5Dset_extent(dset, &written_size);
676
677 // Configure the hyperslab of where to write
678 H5Sclose(ds_file);
679 ds_file = H5Dget_space(dset);
680 H5Sselect_hyperslab(ds_file, H5S_SELECT_SET, &offset, NULL, &i, NULL);
681
682 H5Dwrite(dset, dt_higcell, ds_mem, ds_file, H5P_DEFAULT, buff);
683 }
684 higcit_destroy(iter);
685
686 }
687 H5Dclose(dset);
688 H5Sclose(ds_mem);
689 H5Sclose(ds_file);
690}

692hig_cell *higio_load_tree_hdf5(higio_hdf5* ctx, const char* path)
693{
694 hig_cell *tree = NULL;
695
696 // Open tree dataset
697 hid_t dset = H5Dopen(ctx->fd, path, H5P_DEFAULT);
698 hid_t ds_file = H5Dget_space(dset);
699
700 // Get the chunk size
701 hsize_t chunk_size;
702 {
703 hid_t dset_plist = H5Dget_create_plist(dset);
704 H5Pget_chunk(dset_plist, 1, &chunk_size);
705 H5Pclose(dset_plist);
706 }
707
708 // Get the total size of the tree
709 hsize_t total_size;
710 H5Sget_simple_extent_dims(ds_file, &total_size, NULL);
711
712 if(total_size) {
713 hig_serial_tree buff[chunk_size];
714 hid_t dtype = H5Dget_type(dset);
715 hid_t ds_mem = H5Screate_simple(1, &chunk_size, &chunk_size);
716
717 // Calculate the number of chunks the dataset is split into
718 size_t num_chunks = total_size / chunk_size;
719 if(total_size % chunk_size)
720 ++num_chunks;
721
722 higcit_celliterator *iter = NULL;
723 hig_cell *parent = NULL;
724 size_t numchildren;
725 size_t child_iter;
726
727 hsize_t rem_size = total_size;
728 for(size_t chunk = 0; chunk < num_chunks; ++chunk) {
729 hsize_t offset = total_size - rem_size;
730 hsize_t read_size;
731 if(rem_size >= chunk_size) {
732 // Reading a whole chunk
733 read_size = chunk_size;
734 rem_size -= chunk_size;
735 } else {
736 // Final reading with a smaller chunk
737 read_size = rem_size;
738 H5Sset_extent_simple(ds_mem, 1, &rem_size, &rem_size);
739 rem_size = 0;
740 }
741
742 H5Sselect_hyperslab(ds_file, H5S_SELECT_SET, &offset, NULL, &read_size, NULL);
743 H5Dread(dset, dtype, ds_mem, ds_file, H5P_DEFAULT, buff);
744
745 // Build the root node if this is the first read.
746 size_t i = 0;
747 if(!tree) {
748 tree = hig_create_root(buff[0].lowpoint, buff[0].highpoint);
749 hig_refine_empty(tree, buff[0].numcells);
750
751 iter = higcit_create_breadth_first(tree);
752 parent = tree; //higcit_getcell(iter);
753
754 numchildren = hig_get_number_of_children(tree);
755 child_iter = 0;
756
757 ++i;
758 }
759
760 // Use the loaded nodes to build the tree.
761 for(; i < read_size; ++i) {
762 // Get the node that will receive the loaded data
763 if(child_iter >= numchildren) {
764 do {
765 // Sanity check.
766 if(higcit_isfinished(iter)) {
767 char filename[100];
768
769 H5Fget_name(ctx->fd, filename, 100);
770 fprintf(stderr, "Error: Tree stored in %s:%s is inconsistent: stored nodes does not fit in the claimed tree size.\n", filename, path);
771
772 hig_destroy(tree);
773 tree = NULL;
774 goto clean_and_exit;
775 }
776
777 higcit_nextcell(iter);
778 parent = higcit_getcell(iter);
779 numchildren = hig_get_number_of_children(parent);
780 } while(!numchildren);
781
782 child_iter = 0;
783 }
784
785 hig_cell* target = hig_create_leaf(buff[i].lowpoint, buff[i].highpoint, parent, child_iter++);
786 hig_refine_empty(target, buff[i].numcells);
787 }
788 }
789
790 // Another sanity check.
791 if(rem_size) {
792 char filename[100];
793
794 H5Fget_name(ctx->fd, filename, 100);
795 fprintf(stderr, "Error: Tree stored in %s:%s is inconsistent: there are not enought stored nodes to fill-in the claimed tree size.\n", filename, path);
796
797 hig_destroy(tree);
798 tree = NULL;
799 }
800
801clean_and_exit:
802 higcit_destroy(iter);
803 H5Sclose(ds_mem);
804 H5Tclose(dtype);
805 }
806
807 H5Sclose(ds_file);
808 H5Dclose(dset);
809
810 return tree;
811}

813struct _prop_writer {
814 int rank;
815 hid_t ds_mem, ds_file;
816 hid_t dset;
817 hid_t dtype;
818 hsize_t size[2];
819 hsize_t max_size[2];
820 hsize_t dimension;
821
822 // Stuff to iterate through the leafs of all trees
823 hsize_t offset[2];
824 hid_t datatype;
825};

827static void _prop_writer_init_dataset(higio_hdf5 *ctx, const char *dataset_path,
828 hsize_t max_size, size_t prop_dimension, hid_t datatype,
829 bool use_var_env_compression, struct _prop_writer* wprops)
830{
831 assert(prop_dimension > 0);
832 assert(max_size > 0);
833
834 enum compression_type {
835 DEFAULT,
836 NONE,
837 GZIP,
838 MAFISC,
839 UNKNOWN
840 };
841 static enum compression_type env_comp_type = UNKNOWN;
842
843 wprops->rank = prop_dimension == 1 ? 1 : 2;
844
845 wprops->size[0] = 0;
846 wprops->size[1] = prop_dimension;
847
848 wprops->max_size[0] = max_size;
849 wprops->max_size[1] = prop_dimension;
850
851 memset(wprops->offset, 0, sizeof wprops->offset);
852 wprops->datatype = datatype;
853
854 wprops->ds_file = H5Screate_simple(wprops->rank, wprops->size, wprops->max_size);
855
856 hsize_t csize[2] = {chunk_size, prop_dimension};
857 wprops->ds_mem = H5Screate_simple(wprops->rank, csize, csize);
858
859 hid_t dset_plist = H5Pcreate(H5P_DATASET_CREATE);
860 H5Pset_chunk(dset_plist, wprops->rank, csize);
861
862 enum compression_type used = use_var_env_compression ? env_comp_type : DEFAULT;
863
864 if(used == UNKNOWN) {
865 // Find what compression to use, per environment variable
866 char *choice = getenv("HIGTREE_OUTPUT_COMPRESSION");
867 const char *const choices[] = {
868 "DEFAULT",
869 "NONE",
870 "GZIP",
871 "MAFISC"
872 };
873
874 env_comp_type = DEFAULT;
875 if(choice) {
876 for(size_t i = 1; i < (sizeof(choices)/sizeof(char*)); ++i) {
877 if(strcmp(choices[i], choice) == 0) {
878 env_comp_type = i;
879 break;
880 }
881 }
882 }
883 used = env_comp_type;
884 }
885
886 unsigned int filter_option;
887 const unsigned int GZIP_LEVEL = 4;
888 switch(used) {
889 case DEFAULT:
890 {
891 static enum compression_type available = UNKNOWN;
892 switch(available) {
893 case GZIP:
894 H5Pset_deflate(dset_plist, GZIP_LEVEL);
895 break;
896 case MAFISC:
897 filter_option = 0;
898 H5Pset_filter(dset_plist, (H5Z_filter_t)32002,
899 H5Z_FLAG_OPTIONAL, 1, &filter_option);
900 break;
901 case UNKNOWN:
902 default:
903 // Disable automatic error message to not alarm user...
904 H5Eset_auto(H5E_DEFAULT, NULL, NULL);
905
906 // Try to use MAFISC filter, then fallback to DEFLATE
907 filter_option = 0;
908 if(H5Pset_filter(dset_plist, (H5Z_filter_t)32002,
909 H5Z_FLAG_OPTIONAL, 1, &filter_option) < 0) {
910 int myrank;
911 MPI_Comm_rank(MPI_COMM_WORLD, &myrank);
912 // Use our own, more friendly warning message:
913 if(myrank == 0)
914 fputs("Warning: MAFISC compression filter not found, falling back to GZIP.\n", stderr);
915 H5Pset_deflate(dset_plist, GZIP_LEVEL);
916 available = GZIP;
917 } else {
918 available = MAFISC;
919 }
920
921 H5Eset_auto(H5E_DEFAULT, (H5E_auto_t) H5Eprint, stderr);
922 }
923 }
924 break;
925 case NONE:
926 break;
927 case GZIP:
928 H5Pset_deflate(dset_plist, GZIP_LEVEL);
929 break;
930 case MAFISC:
931 filter_option = 0;
932 H5Pset_filter(dset_plist, (H5Z_filter_t)32002,
933 H5Z_FLAG_OPTIONAL, 1, &filter_option);
934 break;
935 case UNKNOWN:
936 ; // Can't happen.
937 }
938
939 wprops->dset = H5Dcreate(ctx->fd, dataset_path, datatype,
940 wprops->ds_file, H5P_DEFAULT, dset_plist, H5P_DEFAULT);
941 H5Pclose(dset_plist);
942}

944static void _prop_write_chunk(size_t size, void* buff, struct _prop_writer *wprops)
945{
946 hsize_t count[2] = {size, wprops->max_size[1]};
947
948 // Set the size to fit this chunk
949 wprops->size[0] += size;
950 H5Dset_extent(wprops->dset, wprops->size);
951
952 // Set the slice on where to write the chunk
953 H5Sset_extent_simple(wprops->ds_file, wprops->rank, wprops->size, wprops->max_size);
954 H5Sselect_hyperslab(wprops->ds_file, H5S_SELECT_SET,
955 wprops->offset, NULL, count, NULL);
956 wprops->offset[0] += size;
957
958 // Resize the dataspace if last chunk
959 if(size != chunk_size) {
960 H5Sset_extent_simple(wprops->ds_mem, wprops->rank, count, count);
961 }
962
963 // Do the writing
964 H5Dwrite(wprops->dset, wprops->datatype, wprops->ds_mem,
965 wprops->ds_file, H5P_DEFAULT, buff);
966}

971 H5Sclose(wprops->ds_file);
972 H5Dclose(wprops->dset);
973}

1353 /* The corners are enumerated in the following order:
1354 * 2 +--------+ 3
1355 * | |
1356 * | |
1357 * | |
1358 * 0 +--------+ 1
1359 *
1360 * But XDMF expect them in the following order (3 swapped with 2):
1361 *
1362 * 3 +--------+ 2
1363 * | |
1364 * | |
1365 * | |
1366 * 0 +--------+ 1

1369 */

1373 for(iter = sd_get_domain_celliterator(sd);
1374 !higcit_isfinished(iter); higcit_nextcell(iter))
1375 {

1378 hig_get_highpoint(c, lims[0]);
1379 hig_get_lowpoint(c, lims[1]);

1382 for(size_t i = 0; i < PERMS; ++i) {
1383 for(size_t j = 0; j < DIM; ++j) {
1384 p[j] = lims[(i>>j) & 1u][j];
1385 }
1386
1387 gsize curr_id;
1388
1389 gboolean found = g_hash_table_lookup_extended(points,
1390 p, NULL, (gpointer *)&curr_id);
1391 if(!found) {
1392 POINT_ASSIGN(buff_points[idx_points], p);
1393 ++idx_points;
1394 if(idx_points == chunk_size) {
1395 _prop_write_chunk(idx_points, buff_points, &writer_points);
1396 *num_points += idx_points;
1397 idx_points = 0;
1398 }
1399 curr_id = points_ids++;
1400 g_hash_table_insert(points, p, GSIZE_TO_POINTER(curr_id));
1401 p = malloc(sizeof(Point));
1402 }
1403 buff_cells[idx_cells][idxmap[i%4] + 4 * (i/4)] = curr_id;
1404 }
1405 ++idx_cells;
1406 if(idx_cells == chunk_size) {
1407 _prop_write_chunk(idx_cells, buff_cells, &writer_cells);
1408 *num_cells += idx_cells;
1409 idx_cells = 0;
1410 }
1411 }
1412 free(p);
1413 higcit_destroy(iter);
1414 g_hash_table_destroy(points);
1415
1416 // Flush buffer
1417 if(idx_points) {
1418 _prop_write_chunk(idx_points, buff_points, &writer_points);
1419 *num_points += idx_points;
1420 }
1421 if(idx_cells) {
1422 _prop_write_chunk(idx_cells, buff_cells, &writer_cells);
1423 *num_cells += idx_cells;
1424 }

1426 free(buff_cells);
1427 free(buff_points);

1429 _prop_writer_destroy(&writer_points);
1430 _prop_writer_destroy(&writer_cells);
