151static hig_cell *__higcit_get_first_of_bounding_box(hig_cell *cell, higcit_bounding_box_extra *itex) {
152 hig_cell *p = cell;
153 while (p->children != NULL) {
154 itex->curlevel++;
155 int level = itex->curlevel;
156 itex->cells[level] = p;
157 hig_get_cell_coords_of_point(p, itex->lowpoint, itex->ppl[level]);
158 hig_get_cell_coords_of_point(p, itex->highpoint, itex->pph[level]);
159 for(int i = 0; i < DIM; i++) {
160 if (itex->ppl[level][i] < 0) {
161 itex->ppl[level][i] = 0;
162 }
163 if (itex->pph[level][i] >= p->numcells[i]) {
164 itex->pph[level][i] = p->numcells[i] - 1;
165 }
166 }
167 POINT_ASSIGN(itex->pps[level], itex->ppl[level]);
168 int j = hig_toposition(p->numcells, itex->pps[level]);
169 p = p->children[j];
170 }
171 return p;
172}

174static hig_cell *__higcit_bounding_box_inc(higcit_celliterator *it) {
175 higcit_bounding_box_extra *itex = (higcit_bounding_box_extra *) it->ex;
176 it->bi.c = NULL;
177 while (itex->curlevel >= 0) {
178 int level = itex->curlevel;
179 int i = DIM - 1;
180 while (i >= 0 && itex->pps[level][i] >= itex->pph[level][i]) {
181 itex->pps[level][i] = itex->ppl[level][i];
182 i--;
183 }
184 if (i >= 0) {
185 itex->pps[level][i]++;
186 hig_cell *p = itex->cells[level];
187 int j = hig_toposition(p->numcells, itex->pps[level]);
188 p = p->children[j];
189 it->bi.c = __higcit_get_first_of_bounding_box(p, itex);
190 break;
191 } else {
192 itex->curlevel--;
193 }
194 }
195 return it->bi.c;

350static hig_cell * __higcit_breadth_first_inc(higcit_celliterator *it) {
351 higcit_breadth_first_extra *itex = (higcit_breadth_first_extra *) it->ex;
352 hig_cell *p = it->bi.c;
353 if(p == NULL)
354 return NULL;
355
356 // If there are children, there is at least one other
357 // level that must be visited.
358 if(!itex->has_more_level && hig_get_number_of_children(p) > 0) {
359 itex->has_more_level = p;
360 }
361
362 // Now we need to get the next sibling...
363 bool found = false;
364 size_t depth = itex->cur_depth;
365
366 // Outer loop: search for next node in correct level
367 do {
368 // Go right/up in the tree searching for siblings
369 if(p == itex->root)
370 break; // Has no sibling, it is the root!
371 hig_cell *parent = hig_get_parent(p);
372 int nextpos = p->posinparent + 1;
373 if(nextpos < hig_get_number_of_children(parent)) {
374 p = hig_get_child(parent, nextpos);
375 // Found a possible sibiling...
376 if(depth == itex->cur_depth) {
377 found = true;
378 } else if(hig_get_number_of_children(p) > 0) {
379 // Inner loop: Go down in the tree searching for
380 // a node in the correct depth.
381 while(depth < itex->cur_depth) {
382 if(hig_get_number_of_children(p) > 0) {
383 p = hig_get_child(p, 0);
384 ++depth;
385 } else {
386 // Dead end... there is no node
387 // of the required depth on this branch.
388 // break the down search, and continue
389 // to the sibling search.
390 break;
391 }
392 }
393 if(depth == itex->cur_depth)
394 found = true;
395 }
396 // If not found, the next iteration will continue the
397 // search to the next sibling.
398 } else {
399 // No more siblings. Go up in the tree.
400 p = parent;
401 --depth;
402 }
403 } while(!found);
404
405 if(!found) {
406 // Within current depth, no new sibilings were found.
407 // Depth exausted, continue with next level if needed.
408 if(itex->has_more_level) {
409 assert(depth == 0);
410 assert(p == itex->root);
411
412 depth = ++itex->cur_depth;
413 p = hig_get_child(itex->has_more_level, 0);
414 itex->has_more_level = NULL;
415 found = true;
416 }
417 else {
418 // No more levels to seach for, iteration is done.
419 found = false;
420 }
421 }
422
423 if(found) {
424 it->bi.c = p;
425 } else {
426 // All levels exausted, iteration completed.
427 it->bi.c = NULL;
428 }
429
430 return it->bi.c;
431}
