257hig_cell * hig_refine_uniform(hig_cell * cell, int numcells[DIM]) {
258 assert(cell != NULL);
259 assert(cell->children == NULL);
260
261 POINT_ASSIGN(cell->numcells, numcells);
262 Point delta;
263 hig_get_delta(cell, delta);
264 POINT_DIV(delta, delta, cell->numcells);
265 POINT_ASSERT_ALL_TRUE(numcells, > 0);
266
267 int numChildren = hig_get_number_of_children(cell);
268 ALLOC(hig_cell *, cell->children, numChildren);
269
270 /* Zeroing allocated vector because hig_create_leaf()
271 * asserts if it is NULL... */
272 memset(cell->children, 0, numChildren * sizeof *cell->children);
273
274 for (int j = 0; j < numChildren; j++) {
275 int pp[DIM];
276 hig_tobase(j, numcells, pp);
277 Point chlp, chhp;
278 POINT_MULT(chlp, delta, pp);
279 POINT_ADD(chlp, chlp, cell->lowpoint);
280 POINT_ADD(chhp, chlp, delta);
281 hig_create_leaf(chlp, chhp, cell, j);
282 }
283 return cell;
284}

371void hig_get_cell_coords_of_point(hig_cell *cell, const Point point, int pp[DIM]) {
372 assert(cell != NULL);
373 assert(cell->children != NULL);
374
375 POINT_ASSIGN_SCALAR(pp, 0);
376 for (int dim = 0; dim < DIM; dim++) {
377 int first = 0;
378 int last = cell->numcells[dim] - 1;
379 while (last >= first) {
380 pp[dim] = first;
381 if (last == first) {
382 break;
383 } else {
384 int pfirst = hig_toposition(cell->numcells, pp);
385 pp[dim] = last;
386 int plast = hig_toposition(cell->numcells, pp);
387 coordtype delta = (cell->children[plast]->highpoint[dim] - cell->children[pfirst]->lowpoint[dim])/(last - first + 1);
388 int pos = (int)((point[dim] - cell->children[pfirst]->lowpoint[dim])/delta) + first;
389 if (pos > last) {
390 pos = last;
391 }
392 if (pos < first) {
393 pos = first;
394 }
395 pp[dim] = pos;
396 int ppos = hig_toposition(cell->numcells, pp);
397 if (FLT_LE(point[dim], cell->children[ppos]->lowpoint[dim])) {
398 last = pos - 1;
399 } else if (FLT_GT(point[dim], cell->children[ppos]->highpoint[dim])) {
400 first = pos + 1;
401 } else {
402 break;
403 }
404 }
405 }
406 }
407 for (int dim = 0; dim < DIM; dim++) {
408 if (FLT_LT(point[dim], cell->lowpoint[dim])) {
409 pp[dim] = -1;
410 }
411 if (FLT_GT(point[dim], cell->highpoint[dim])) {
412 pp[dim] = cell->numcells[dim];
413 }
414 }

516bool hig_check_conformable(hig_cell *cell) {
517 if (cell == NULL) {
518 return 1;
519 }
520 if (cell->children == NULL) {
521 return 1;
522 }
523 int numChildren = hig_get_number_of_children(cell);
524 for (int i = 0; i < DIM; i++) {
525 if (FLT_NE(cell->lowpoint[i], cell->children[0]->lowpoint[i])) {
526 return 0;
527 }
528 if (FLT_NE(cell->highpoint[i], cell->children[numChildren-1]->highpoint[i])) {
529 return 0;
530 }
531 }
532 int pp[DIM];
533 for (int j = 0; j < numChildren; j++) {
534 hig_tobase(j, cell->numcells, pp);
535 for (int i = 0; i < DIM; i++) {
536 if (pp[i] < cell->numcells[i] - 1) {
537 int pp2[DIM];
538 POINT_ASSIGN(pp2, pp);
539 pp2[i]++;
540 int j2 = hig_toposition(cell->numcells, pp2);
541 if (FLT_NE(cell->children[j]->highpoint[i], cell->children[j2]->lowpoint[i])) {
542 return 0;
543 }
544 }
545 }
546 }
547 for (int j = 0; j < numChildren; j++) {
548 if (!hig_check_conformable(cell->children[j])) {
549 return 0;
550 }
551 }
552 return 1;
553}

592void hig_adjust_facet_ids(hig_cell *root) {
593 higcit_celliterator *it;
594 for(it = higcit_create_all_leaves(root); !higcit_isfinished(it); higcit_nextcell(it)) {
595 hig_cell *c = higcit_getcell(it);
596 Point cl, ch, ccenter;
597 hig_get_lowpoint(c, cl);
598 hig_get_highpoint(c, ch);
599 hig_get_center(c, ccenter);
600 for(int dim = 0; dim < DIM; dim++) {
601 Point ofcenter;
602 POINT_ASSIGN(ofcenter, ccenter);
603 for(int dir = 0; dir < 2; dir++) {
604 if(dir == 0) {
605 ofcenter[dim] = cl[dim]-EPSDELTA;
606 } else {
607 ofcenter[dim] = ch[dim]+EPSDELTA;
608 }
609 bool isoppositefacetdifferent = false;
610 bool isoppositefacetsmaller = false;
611 hig_cell *ofc = hig_get_cell_with_point(root, ofcenter);
612 if (ofc != NULL) {
613 Point ofl, ofh;
614 hig_get_lowpoint(ofc, ofl);
615 hig_get_highpoint(ofc, ofh);
616 for(int k = 0; k < DIM; k++) {
617 if (k != dim) {
618 if (FLT_NE(cl[k], ofl[k]) || FLT_NE(ch[k], ofh[k])) {
619 isoppositefacetdifferent = true;
620 if (FLT_GT((ch[k] - cl[k]), (ofh[k] - ofl[k]))) {
621 isoppositefacetsmaller = true;
622 }
623 }
624 }
625 }
626 }
627 if (isoppositefacetdifferent) {
628 if (isoppositefacetsmaller) {
629 hig_set_fid_with_dim_dir(c, dim, dir, 0);
630 }
631 } else if (ofc != NULL) {
632 if (dir == 0) {
633 hig_set_fid_with_dim_dir(c, dim, dir, - hig_get_fid_with_dim_dir(ofc, dim, 1-dir));
634 }
635 }
636 }
637 }
638 }
639 higcit_destroy(it);
640}

710int hig_get_facet_with_point(hig_cell *root, int dim, const Point x, hig_facet *facet) {
711 hig_cell *c = hig_get_cell_with_point(root, x);
712 if (c == NULL) {
713 return 0;
714 }
715 Point lowpoint, highpoint;
716 hig_get_lowpoint(c, lowpoint);
717 hig_get_highpoint(c, highpoint);
718 int dir = -1;
719 if (FLT_EQ(x[dim], lowpoint[dim])) {
720 dir = 0;
721 } else if (FLT_EQ(x[dim], highpoint[dim])) {
722 dir = 1;
723 }
724 if (dir == -1) {
725 return 0;
726 }
727 int id = hig_get_fid_with_dim_dir(c, dim, dir);
728 if (id > 0) {
729 facet->c = c;
730 facet->dim = dim;
731 facet->dir = dir;
732 return 1;
733 } else {
734 Point x2;
735 POINT_ASSIGN(x2, x);
736 if (dir == 0) {
737 x2[dim] -= EPSDELTA;
738 } else {
739 x2[dim] += EPSDELTA;
740 }
741 hig_cell *c2 = hig_get_cell_with_point_from_another_cell(c, x2);
742 if (c2 == NULL) {
743 return 0;
744 }
745 facet->c = c2;
746 facet->dim = dim;
747 facet->dir = 1-dir;
748 return 1;
749 }
750 return 0;
751}
