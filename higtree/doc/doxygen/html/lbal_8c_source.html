281static bool _try_merge_chunk_into_tree(Point lowpoint, Point highpoint, hig_cell **rcells,
282 unsigned *numcells, hig_cell *tree)
283{
284 int matchdim = -1;
285
286 // For each dimension, try to find one where this chunk matches with the tree
287 for(unsigned dim = 0; dim < DIM; ++dim)
288 {
289 if(POS_EQ(tree->lowpoint[dim], highpoint[dim])
290 || POS_EQ(tree->highpoint[dim], lowpoint[dim]))
291 {
292 // Found a candidate, assert that all other sizes and dimensions matches
293 for(unsigned odim = 0; odim < DIM; ++odim) {
294 if(dim == odim)
295 continue;
296
297 if(POS_NE(highpoint[odim], tree->highpoint[odim])
298 || POS_NE(lowpoint[odim], tree->lowpoint[odim])
299 || numcells[odim] != tree->numcells[odim])
300 return false;
301 }
302 matchdim = dim;
303
304 // Inneficient goto substitute:
305 break;
306 }
307 }
308
309 // Inneficient goto label substitute:
310 if(matchdim == -1)
311 return false;
312
313 // Calculate the new dimension
314 unsigned newncells[DIM];
315 memcpy(newncells, numcells, sizeof newncells);
316 newncells[matchdim] += tree->numcells[matchdim];
317
318 unsigned total_size = 1;
319 for(unsigned dim = 0; dim < DIM; ++dim) {
320 total_size *= newncells[dim];
321 }
322
323 // Order the chunks...
324 hig_cell **lower;
325 int *lower_numcells;
326
327 hig_cell **higher;
328 int *higher_numcells;
329
330 unsigned middle;
331 if(POS_EQ(tree->lowpoint[matchdim], highpoint[matchdim])) {
332 // The tree comes after the new chunk
333 lower = rcells;
334 lower_numcells = (int*)numcells;
335 higher = tree->children;
336 higher_numcells = tree->numcells;
337 middle = numcells[matchdim];
338
339 // Adjust tree's lowpoint
340 tree->lowpoint[matchdim] = lowpoint[matchdim];
341 } else {
342 // The new chunk comes after the tree
343 lower = tree->children;
344 lower_numcells = tree->numcells;
345 higher = rcells;
346 higher_numcells = (int*)numcells;
347 middle = tree->numcells[matchdim];
348
349 // Adjust tree's highpoint
350 tree->highpoint[matchdim] = highpoint[matchdim];
351 }
352
353 // Merge the nodes
354 DECL_AND_ALLOC(hig_cell *, children, total_size);
355 int pp[DIM] = {0};
356 for(unsigned i = 0; i < total_size; ++i) {
357 // Find the destination idx
358 int idx = hig_toposition((int *)newncells, pp);
359
360 // Assign the element to its new position
361 if(pp[matchdim] < middle) {
362 int s_idx = hig_toposition(lower_numcells, pp);
363 children[idx] = lower[s_idx];
364 } else {
365 int npp[DIM];
366 memcpy(npp, pp, sizeof pp);
367 npp[matchdim] -= middle;
368
369 int s_idx = hig_toposition(higher_numcells, npp);
370 children[idx] = higher[s_idx];
371 }
372
373 // Adjust children to tree
374 children[idx]->posinparent = idx;
375 children[idx]->parent = tree;
376
377 // Increment element's position
378 pp[0]++;
379 for(unsigned dim = 0; dim < DIM && pp[dim] == newncells[dim];) {
380 pp[dim] = 0;
381 ++pp[++dim];
382 }
383 }
384
385 // Adjust the tree's root node
386 free(tree->children);
387 tree->children = children;
388
389 tree->numcells[matchdim] = newncells[matchdim];
390
391 return true;
392}

414static struct _lb_output_tree *
415_try_merge_tree_list(
416 size_t num_merged, struct _lb_output_tree *merged,
417 size_t num_unmerged, struct _lb_output_tree *unmerged,
418 size_t *ret_count
419 )
420{
421 const size_t maximum_count = num_merged + num_unmerged;
422 DECL_AND_ALLOC(struct _lb_output_tree, next_merged, maximum_count);
423 size_t next_merged_count = 0;
424
425 size_t next_unmerged_count = 0;
426
427 // Try to match each merged with each unmerged
428 for(size_t i = 0; i < num_merged; ++i) {
429 for(size_t j = next_unmerged_count; j < num_unmerged; ++j) {
430 bool merge_res = _try_merge_trees(&merged[i], &unmerged[j]);
431 if(merge_res) {
432 // Put all the newly created trees at the begining of the
433 // unmerged vector, swapping places the just merged tree
434 unmerged[j] = unmerged[next_unmerged_count];
435 unmerged[next_unmerged_count] = merged[i];
436 ++next_unmerged_count;
437
438 // Fill the hole with the last element in vector
439 merged[i] = merged[--num_merged];
440
441 break;
442 }
443 }
444 }
445
446 // Try to match each unmerged with itself
447 while(next_unmerged_count < num_unmerged) {
448 bool has_merged = false;
449 for(size_t j = next_unmerged_count+1; j < num_unmerged; ++j) {
450 has_merged = _try_merge_trees(&unmerged[next_unmerged_count],
451 &unmerged[j]);
452 if(has_merged) {
453 ++next_unmerged_count;
454 unmerged[j] = unmerged[--num_unmerged];
455
456 //goto continue_while;
457 break;
458 }
459 }
460
461 // Poor goto replacement
462 if(!has_merged) {
463 // The tree has been tried agains all input trees
464 next_merged[next_merged_count++] = unmerged[next_unmerged_count];
465 unmerged[next_unmerged_count] = unmerged[--num_unmerged];
466 }
467
468 //continue_while:;
469 }
470
471 // Put all merged in a single list
472 memcpy(&next_merged[next_merged_count], merged, num_merged * sizeof *merged);
473 next_merged_count += num_merged;
474 next_merged = realloc(next_merged, next_merged_count * sizeof *next_merged);
475
476 // Test if recursion end
477 if(num_unmerged) {
478 struct _lb_output_tree *ret = _try_merge_tree_list(next_merged_count, next_merged,
479 next_unmerged_count, unmerged, ret_count);
480 free(next_merged);
481 return ret;
482 } else {
483 *ret_count = next_merged_count;
484 return next_merged;
485 }

494
495 // Uniquely enumerate all trees taking part in the partitioning
496 {
497 MPI_Request reqs[group_size - rank - 1];
498
499 for(int i = rank+1; i < group_size; ++i) {
500 MPI_Isend(&ctx->num_in_higs, 1, MPI_UNSIGNED, i, 73, ctx->group, &reqs[i - rank - 1]);
501 }

655 // a) count and send the number of elements per tree the remote process
656 // will receive;

666 // Fill in and send the buffers
667 MPI_Request *size_send_reqs = &send_reqs[0];
668 MPI_Request *tree_send_reqs = &send_reqs[num_exp_procs];
669 MPI_Request *groups_send_reqs = &send_reqs[2 * num_exp_procs];
670
671 for(unsigned pidx = 0; pidx < num_exp_procs; ++pidx) {
672 // Allocate the buffers to hold the data to send to pidx:
673 DECL_AND_ALLOC(hig_serial_tree, buf, num_nodes_to_send[pidx]);
674 tree_buffs[pidx] = buf;
675
676 ALLOC_INFER(sizes_buffs[pidx], num_trees_to_send[pidx] * DIM);
677 memset(sizes_buffs[pidx], 0, num_trees_to_send[pidx] * DIM * sizeof *sizes_buffs[pidx]);
678 ALLOC_INFER(groups_buffs[pidx], num_trees_to_send[pidx]);
679
680 // Serialize each tree node outgoing to pidx
681 int last_tree = -1;
682 int first_tree_child = -1;
683 int p = all_exp_procs[pidx];
684 unsigned otree_it = 0;
685 unsigned tree_child_count;
686 while(buf - tree_buffs[pidx] < num_nodes_to_send[pidx]) {
687 // Get the current element to export
688 struct _to_export_elem* to_exp = &to_export[k++];
689 assert(to_exp->proc == p);
690
691 // Get its parent tree
692 size_t ptree_idx = to_exp->fid[0] - ctx->first_tree_id;
693 hig_cell *ptree = params.tree_index[ptree_idx]->tree;
694
695 // Get the subtree from the parent
696 hig_cell *elem = hig_get_child(ptree, to_exp->fid[1]);
697
698 // Serialize it
699 buf += hs_serialize(buf, elem);
700 assert(num_nodes_to_send[pidx] >= buf - tree_buffs[pidx]);
701
702 // For each different tree chunk, increment the counter and
703 // find its bounding box
704 if(ptree_idx != last_tree) {
705 if(first_tree_child != -1) {
706 unsigned count;
707 count = _calc_chunk_numcells(
708 params.tree_index[last_tree]->tree,
709 first_tree_child, to_export[k-2].fid[1],
710 &sizes_buffs[pidx][DIM * otree_it++]);
711 assert(count == tree_child_count);
712 }
713 tree_child_count = 1;
714 first_tree_child = to_exp->fid[1];
715
716 // Fill-in the group of this tree:
717 groups_buffs[pidx][otree_it] =
718 params.tree_index[ptree_idx]->group;
719
720 last_tree = ptree_idx;
721 } else {
722 ++tree_child_count;
723 }
724 }
725 if(first_tree_child != -1) {
726 size_t last_tree_child = to_export[k-1].fid[1];
727 last_tree = to_export[k-1].fid[0] - ctx->first_tree_id;
728 unsigned count;
729 count = _calc_chunk_numcells(params.tree_index[last_tree]->tree,
730 first_tree_child, last_tree_child,
731 &sizes_buffs[pidx][DIM * otree_it++]);
732 assert(count == tree_child_count);
733 }
734 assert(otree_it == num_trees_to_send[pidx]);
735
736 // Send the actual tree nodes
737 MPI_Isend(tree_buffs[pidx], num_nodes_to_send[pidx] * sizeof *buf,
738 MPI_BYTE, p, 465, ctx->group, &tree_send_reqs[pidx]);
739
740 // Send the group and the size of each tree
741 MPI_Isend(sizes_buffs[pidx], num_trees_to_send[pidx] * DIM,
742 MPI_UNSIGNED, p, 546, ctx->group, &size_send_reqs[pidx]);
743 MPI_Isend(groups_buffs[pidx], num_trees_to_send[pidx], MPI_UNSIGNED,
744 p, 654, ctx->group, &groups_send_reqs[pidx]);
745
746 assert(num_nodes_to_send[pidx] == buf - tree_buffs[pidx]);
747 }
748 free(all_exp_procs);
749
750 // We now iterate through input trees and build the output trees vector
751 // for this process, removing sent elements, and taking into consideration
752 // if trees must be reused or copyied.
753
754 // Reorder the elements to being exported, this time, disregarding
755 // destination process
756 qsort(to_export, num_exp, sizeof *to_export,
757 (int (*)(const void *, const void *))_just_id_cmp);
758
759 k = 0; // Reseting iterator
760
761 // Handle each input tree
762 for(unsigned i = 0; i < ctx->num_in_higs; ++i)
763 {
764 hig_cell *tree = params.tree_index[i]->tree;
765
766 // Get the number of original elements
767 unsigned numoch = hig_get_number_of_children(tree);
768 bool managed = params.tree_index[i]->managed;
769 hig_cell **nodes;
770 if(managed)
771 nodes = tree->children;
772 else {
773 ALLOC(hig_cell *, nodes, numoch);
774 }
775
776 // Iterate through each exported element, checking each against
777 // the corresponding node in tree, to remove the exported ones
778 struct _to_export_elem* to_exp = &to_export[k];
779 unsigned new_count = 0;
780 for(unsigned j = 0; j < numoch; ++j) {
781 assert(k <= num_exp);
782
783 // If we are not over the exported elements,
784 // then we aren't ahead of its tree
785 assert(k == num_exp
786 || to_exp->fid[0] - ctx->first_tree_id >= i);
787
788 // It we are not over the exported elements,
789 // and if the exported element is in this tree,
790 // then j hasn't passed it yet:
791 assert(k == num_exp
792 || to_exp->fid[0] - ctx->first_tree_id != i
793 || j <= to_exp->fid[1]);
794
795 if(k >= num_exp
796 || to_exp->fid[0] - ctx->first_tree_id != i
797 || j != to_exp->fid[1])
798 {
799 // Node remains in tree...
800 hig_cell *node = hig_get_child(tree, j);
801
802 if(!managed) {
803 hig_cell *tmp = hig_clone(node);
804 tmp->posinparent = node->posinparent;
805 node = tmp;
806 }
807
808 nodes[new_count++] = node;
809 } else {
810 // Node is removed from the tree...
811 if(managed) {
812 hig_cell *node = hig_get_child(tree, j);
813 hig_destroy(node);
814 }
815
816 // Advance exported node iterator
817 to_exp = &to_export[++k];
818 }
819 }
820 nodes = realloc(nodes, new_count * sizeof *nodes);
821
822 // Adapt the actual hig_cell structure to fit the changes
823 if(new_count == 0) {
824 // No elements left, no tree should be output for this one
825 if(managed)
826 // Can't use hig_destroy here because the tree->children
827 // is now invalid, it has been freed by the realloc
828 // just above, and all children have been destroyed in
829 // the loop.
830 free(tree);
831 else
832 // If not managed and has no tree to output, do nothing.
833 ;
834 } else {
835 // There are some remaining elements on this tree,
836 // build the output tree.
837 hig_cell *out;
838 if(managed) {
839 out = tree;
840 // The pointer has been realloc'ed and might
841 // have changed
842 tree->children = nodes;
843 } else {
844 Point p = {0};
845 out = hig_create_root(p, p);
846 out->children = nodes;
847 for(unsigned i = 0; i < new_count; ++i)
848 {
849 nodes[i]->parent = out;
850 }
851 }
852
853 // Find the dimensional size of the new tree,
854 // relies on the information of the old tree
855 unsigned calculated_size;
856 calculated_size = _calc_chunk_numcells(tree, nodes[0]->posinparent, nodes[new_count - 1]->posinparent, (unsigned int*)out->numcells);
857 assert(calculated_size == new_count);
858
859 // Adjust posinparent for children:
860 for(unsigned i = 0; i < new_count; ++i) {
861 nodes[i]->posinparent = i;
862 }
863
864 // Adjust new bounding box
865 _calc_boundingbox_from_children(out);
866
867 // Add the new tree in the output list
868 ctx->output[ctx->num_out_higs].group = params.tree_index[i]->group;
869 ctx->output[ctx->num_out_higs].tree = out;
870
871 ctx->num_out_higs++;
872 }
873 } // end of "for each input tree"
874 free(to_export);
875 }
876
877 // Try to coalesce the trees already set to output
878 size_t output_size;
879 {
880 struct _lb_output_tree *new_output;
881 new_output = _try_merge_tree_list(0, NULL, ctx->num_out_higs, ctx->output,
882 &output_size);
883 free(ctx->output);
884 ctx->output = new_output;
885 ctx->num_out_higs = output_size;
886 }
887
888 // Input tree data is no longer needed from now on
889 free(params.tree_index);
890 ctx->num_in_higs = 0;
891 while(ctx->trees) {
892 void *tmp = ctx->trees;
893 ctx->trees = ctx->trees->next;
894 free(tmp);
895 }
896
897 // Creates a list to be filled with trees that may still be merged.
898 // Use an optimistically allocated size.
899 size_t mergeable_size = 2 * output_size + 1;
900 size_t mergeable_count = 0;
901 DECL_AND_ALLOC(struct _lb_output_tree, mergeable, mergeable_size);
902
903 // Receive the elements from the remote processes
904 size_t recv_elem_buff_size = 0;
905 size_t recv_sizes_buff_size = 0;
906 unsigned *sizes_buf = NULL;
907 unsigned *groups_buf = NULL;
908 hig_serial_tree *elem_buf = NULL;
909 for(int pidx = 0; pidx < num_imp_procs; ++pidx) {
910 // Receive the size of the tree_buf
911 MPI_Status status;
912 MPI_Probe(MPI_ANY_SOURCE, 465, ctx->group, &status);
913 int elem_count = 0;
914 MPI_Get_count(&status, MPI_BYTE, &elem_count);
915
916 // Adjust tree_buf size
917 if(elem_count > recv_elem_buff_size) {
918 recv_elem_buff_size = elem_count;
919
920 // won't use realloc here, because I am not
921 // interested in keeping old data in the buffer
922 free(elem_buf);
923 elem_buf = malloc(elem_count);
924 }
925
926 // Now we asynchronously receive the trees' elements,
927 // while synchronously receiving trees sizes.
928
929 // The receiving of the trees' nodes:
930 MPI_Request elems_req;
931 MPI_Irecv(elem_buf, elem_count, MPI_BYTE, status.MPI_SOURCE, 465,
932 ctx->group, &elems_req);
933
934 // Adjust the tree_count: from bytes to number of elements:
935 assert((elem_count % sizeof *elem_buf) == 0);
936 elem_count /= sizeof *elem_buf;
937
938 // Sync probe for the message of number of elements per tree:
939 MPI_Probe(status.MPI_SOURCE, 546, ctx->group, &status);
940 int sizes_count;
941 MPI_Get_count(&status, MPI_UNSIGNED, &sizes_count);
942
943 // Adjust sizes_buf to fit the data:
944 if(sizes_count > recv_sizes_buff_size) {
945 recv_sizes_buff_size = sizes_count;
946
947 free(sizes_buf);
948 ALLOC_INFER(sizes_buf, sizes_count);
949
950 free(groups_buf);
951 ALLOC_INFER(groups_buf, sizes_count / DIM);
952 }
953
954 // Async receive the groups each incoming tree belongs to:
955 MPI_Request groups_req;
956 MPI_Irecv(groups_buf, sizes_count / DIM, MPI_UNSIGNED,
957 status.MPI_SOURCE, 654, ctx->group, &groups_req);
958
959 // Sync receive the tree chunk dimmensions.
960 MPI_Recv(sizes_buf, sizes_count, MPI_UNSIGNED, status.MPI_SOURCE,
961 546, ctx->group, MPI_STATUS_IGNORE);
962
963 // Adjust sizes_count:
964 assert(sizes_count % DIM == 0);
965 sizes_count /= DIM;
966
967 MPI_Wait(&elems_req, MPI_STATUS_IGNORE);
968 MPI_Wait(&groups_req, MPI_STATUS_IGNORE);
969
970 // Create a buffer to hold temporarily the deserialized
971 // tree nodes.
972 hig_cell **rcells = NULL;
973 size_t rcells_size = 0;
974
975 // Deserialize one tree at a time
976 hig_serial_tree *elemptr = elem_buf;
977 for(size_t tree = 0; tree < sizes_count; ++tree) {
978 size_t tree_size = 1;
979 for(unsigned j = 0; j < DIM; ++j) {
980 tree_size *= sizes_buf[tree * DIM + j];
981 }
982 if(rcells_size < tree_size) {
983 free(rcells);
984 rcells_size = tree_size;
985 ALLOC(hig_cell *, rcells, tree_size);
986 }
987
988 elemptr += hs_deserialize(&rcells[0], elemptr);
989 Point highpoint, lowpoint;
990 hig_get_highpoint(rcells[0], highpoint);
991 hig_get_lowpoint(rcells[0], lowpoint);
992
993 for(size_t i = 1; i < tree_size; ++i) {
994 elemptr += hs_deserialize(&rcells[i], elemptr);
995 Point chigh, clow;
996 hig_get_highpoint(rcells[i], chigh);
997 hig_get_lowpoint(rcells[i], clow);
998
999 for(size_t dim = 0; dim < DIM; ++dim) {
1000 if(chigh[dim] > highpoint[dim])
1001 highpoint[dim] = chigh[dim];
1002 if(clow[dim] < lowpoint[dim])
1003 lowpoint[dim] = clow[dim];
1004 }
1005 }
1006
1007 // Try to integrate the new tree into each of the already
1008 // present trees:
1009 bool merged = false;
1010 for(unsigned i = 0; i < ctx->num_out_higs; ++i) {
1011 merged = groups_buf[tree] == ctx->output[i].group
1012 && _try_merge_chunk_into_tree(lowpoint, highpoint, rcells,
1013 &sizes_buf[tree * DIM], ctx->output[i].tree);
1014 if(merged) {
1015 // The new merged tree may now be merged again...
1016 // Add to the corresponding list
1017 assert(mergeable_count <= mergeable_size);
1018 if(mergeable_count >= mergeable_size) {
1019 mergeable_size = 1 + mergeable_size * 2;
1020 mergeable = realloc(mergeable,
1021 mergeable_size * sizeof *mergeable);
1022 }
1023 mergeable[mergeable_count++] = ctx->output[i];
1024 ctx->output[i] = ctx->output[--ctx->num_out_higs];
1025 break;
1026 }
1027 }
1028
1029 if(!merged) {
1030 // If couldn't merge, create a new output tree:
1031 hig_cell *ntree = hig_create_root(lowpoint, highpoint);
1032 hig_refine_empty(ntree, (int *)&sizes_buf[tree * DIM]);
1033
1034 for(unsigned i = 0; i < tree_size; ++i) {
1035 rcells[i]->posinparent = i;
1036 rcells[i]->parent = ntree;
1037 ntree->children[i] = rcells[i];
1038 }
1039
1040 // Add the new tree to the output list
1041 if(ctx->num_out_higs >= output_size) {
1042 output_size = 1 + output_size * 2;
1043 ctx->output = realloc(ctx->output,
1044 output_size * sizeof *ctx->output);
1045 }
1046 ctx->output[ctx->num_out_higs].tree = ntree;
1047 ctx->output[ctx->num_out_higs].group = groups_buf[tree];
1048 ctx->num_out_higs++;
1049 }
1050 }
1051 free(rcells);
1052 }
1053 free(sizes_buf);
1054 free(elem_buf);
1055 free(groups_buf);
1056
1057 // Final merge of all output trees
1058 {
1059 struct _lb_output_tree *new_output;
1060 new_output = _try_merge_tree_list(ctx->num_out_higs, ctx->output,
1061 mergeable_count, mergeable, &output_size);
1062 free(ctx->output);
1063 free(mergeable);
1064
1065 ctx->output = new_output;
1066 ctx->num_out_higs = output_size;
1067 }
1068
1069 // Free the sending buffers
1070 MPI_Waitall((sizeof send_reqs) / (sizeof send_reqs[0]), send_reqs,
1071 MPI_STATUSES_IGNORE);
1072
1073 for(int p = 0; p < num_exp_procs; ++p) {
1074 free(sizes_buffs[p]);
1075 free(tree_buffs[p]);
1076 free(groups_buffs[p]);
1077 }
1078
1079 assert(output_size >= ctx->num_out_higs);
1080 ctx->output = realloc(ctx->output, ctx->num_out_higs * sizeof *ctx->output);
1081 }
1082 Zoltan_Destroy(&zz);
1083}

1107static struct _proc_conectivity*
1108_proc_conn_parallel_find(load_balancer *ctx)
1109{
1110 DECL_AND_ALLOC(struct _proc_conectivity, pc, 1);
1111
1112 if(DIM <= 5)
1113 // relativelly small number of possible neighbors
1114 pc->max_neighbors = 2 * upow(3, DIM);
1115 else
1116 // number would be too big, just use some arbitrary value
1117 pc->max_neighbors = 600;
1118
1119 ALLOC_INFER(pc->neighbors, pc->max_neighbors);
1120 pc->num_neighbors = 0;
1121
1122 MPI_Comm comm;
1123
1124 // Take those procs who have no tress out of computation
1125 if(!ctx->num_out_higs) {
1126 MPI_Comm_split(ctx->group, MPI_UNDEFINED, 0, &comm);
1127 return pc;
1128 }
1129
1130 int outer_rank;
1131 MPI_Comm_rank(ctx->group, &outer_rank);
1132 MPI_Comm_split(ctx->group, 0, outer_rank, &comm);
1133
1134 int ntasks;
1135 MPI_Comm_size(comm, &ntasks);
1136
1137 // Find local bounding box
1138 Rect lbounds;
1139 hig_get_bounding_box(ctx->output[0].tree, &lbounds);
1140 for(unsigned i = 1; i < ctx->num_out_higs; ++i) {
1141 Rect tbounds;
1142 hig_get_bounding_box(ctx->output[i].tree, &tbounds);
1143
1144 for(unsigned dim = 0; dim < DIM; ++dim) {
1145 if(lbounds.lo[dim] > tbounds.lo[dim])
1146 lbounds.lo[dim] = tbounds.lo[dim];
1147
1148 if(lbounds.hi[dim] < tbounds.hi[dim])
1149 lbounds.hi[dim] = tbounds.hi[dim];
1150 }
1151 }
1152
1153 // Gather all procs outer ranks
1154 DECL_AND_ALLOC(int, proc_ranks, ntasks);
1155 MPI_Allgather(&outer_rank, 1, MPI_INT, proc_ranks, 1, MPI_INT, comm);
1156
1157 // Gather all procs bounding boxes
1158 DECL_AND_ALLOC(Rect, rbounds, ntasks);
1159 MPI_Allgather(&lbounds, sizeof(Rect), MPI_BYTE, rbounds,
1160 sizeof(Rect), MPI_BYTE, comm);
1161
1162 // Find the intersection between this process and every other
1163 for(int i = 0; i < ntasks; ++i) {
1164 if(outer_rank == proc_ranks[i])
1165 // Skip if the current process...
1166 continue;
1167
1168 Rect isect;
1169 if(rect_intersect(&rbounds[i], &lbounds, &isect))
1170 _proc_conn_add_touching(pc, proc_ranks[i], &isect);
1171 }
1172
1173 free(rbounds);
1174 free(proc_ranks);
1175 MPI_Comm_free(&comm);
1176
1177 return pc;
1178}

1180static void _local_connectivity_append_tree_neighbor(struct _local_connectivity *lc,
1181 size_t tidx_orig, size_t tidx_dest, Rect *isect_orig, Point displacement)
1182{
1183 // Augment local conectivity with relevant
1184 // tree neighbors.
1185 const size_t nbs_count =
1186 ++lc->tree_neighbors[tidx_orig].num_neighbors;
1187
1188 REALLOC_INFER(lc->tree_neighbors[tidx_orig].neighbors, nbs_count);
1189 struct _local_neighbor *new =
1190 &lc->tree_neighbors[tidx_orig].neighbors[nbs_count-1];
1191
1192 new->intersection = *isect_orig;
1193 new->neighbor_tree = tidx_dest;
1194 POINT_ASSIGN(new->displacement, displacement);
1195}

1197struct _local_connectivity *
1198_local_connectivity_build(load_balancer *ctx, struct _local_portals *lps)
1199{
1200 DECL_AND_ALLOC(struct _local_connectivity, lc, 1);
1201 ALLOC_INFER(lc->tree_neighbors, ctx->num_out_higs);
1202 memset(lc->tree_neighbors, 0, ctx->num_out_higs * sizeof *lc->tree_neighbors);
1203 lc->num_trees = ctx->num_out_higs;
1204
1205 // Find geometric contact between trees
1206 {
1207 struct _local_neighbor curr[ctx->num_out_higs];
1208 unsigned curr_size = 0;
1209 for(unsigned i = 0; i < ctx->num_out_higs; ++i) {
1210 // If "i" doesn't have fringe, skip it
1211 if(!ctx->group_fringe[ctx->output[i].group])
1212 continue;
1213
1214 for(unsigned j = 0; j < ctx->num_out_higs; ++j) {
1215 // If "i" and "j" have different groups, skip "j".
1216 if(ctx->output[i].group != ctx->output[j].group || i == j)
1217 continue;
1218
1219 Rect a, b;
1220 hig_get_bounding_box(ctx->output[i].tree, &a);
1221 hig_get_bounding_box(ctx->output[j].tree, &b);
1222 if(rect_intersect(&a, &b, &curr[curr_size].intersection)) {
1223 curr[curr_size].neighbor_tree = j;
1224 POINT_ASSIGN_SCALAR(curr[curr_size].displacement, 0);
1225 ++curr_size;
1226 }
1227 }
1228 if(curr_size) {
1229 ALLOC_INFER(lc->tree_neighbors[i].neighbors, curr_size);
1230 memcpy(lc->tree_neighbors[i].neighbors, curr,
1231 curr_size * sizeof *curr);
1232 lc->tree_neighbors[i].num_neighbors = curr_size;
1233
1234 curr_size = 0;
1235 }
1236 }
1237 assert(curr_size == 0);
1238 }
1239
1240 // Find portal contact between trees.
1241 for(struct _local_portal *iter = lps->start; iter; iter = iter->next) {
1242 for(struct _portal_tree_isect *i0 = iter->side_isect[0]; i0; i0 = i0->next) {
1243 for(struct _portal_tree_isect *i1 = iter->side_isect[1]; i1; i1 = i1->next) {
1244 // Move side 1 intersection to postion of side 0:
1245 // Step 1: calculate the displacement
1246 Point delta;
1247 POINT_SUB(delta, iter->p->r[0].lo, iter->p->r[1].lo);
1248 // Step 2: apply the displacement to the side 1 intersection
1249 Rect s1;
1250 rect_translate(delta, &i1->isect, &s1);
1251
1252 // Test for intersection:
1253 Rect isect_s0;
1254 if(rect_intersect(&i0->isect, &s1, &isect_s0)) {
1255 // Warp inside the same process found!
1256
1257 Point opposite_delta;
1258 POINT_MULT_SCALAR(opposite_delta, delta, -1);
1259
1260 // Calculate the contact area on the side 1:
1261 Rect isect_s1;
1262 rect_translate(opposite_delta, &isect_s0, &isect_s1);
1263
1264 // Augment local conectivity with relevant
1265 // tree neighbors.
1266 // On side 0:
1267 _local_connectivity_append_tree_neighbor(lc,
1268 i0->tree_idx, i1->tree_idx, &isect_s0,
1269 opposite_delta);
1270 // On side 1:
1271 _local_connectivity_append_tree_neighbor(lc,
1272 i1->tree_idx, i0->tree_idx, &isect_s1,
1273 delta);
1274 }
1275 }
1276 }
1277 }
1278
1279 return lc;
1280}

1351struct _local_portals* _local_portals_create(load_balancer *ctx)
1352{
1353 DECL_AND_ALLOC(struct _local_portals, ret, 1);
1354 ret->local_portal_count = 0;
1355 ret->start = NULL;
1356
1357 struct _local_portal new_lp;
1358 memset(&new_lp, 0, sizeof new_lp);
1359
1360 // Empty struct to be filled
1361 struct _portal_tree_isect new_isect;
1362 memset(&new_isect, 0, sizeof new_isect);
1363
1364 // For each portal, test every local tree.
1365 size_t portal_gid = 0;
1366 for(struct _lb_portal *cur = ctx->portals; cur; cur = cur->next) {
1367
1368 bool isect_found = false;
1369 for(size_t tidx = 0; tidx < ctx->num_out_higs; ++tidx) {
1370 if(ctx->output[tidx].group != cur->group) {
1371 continue;
1372 }
1373
1374 // Get tree bounding box.
1375 hig_cell *t = ctx->output[tidx].tree;
1376 Rect bbox;
1377 hig_get_bounding_box(t, &bbox);
1378
1379 // Test if each side of the portal intersects the tree.
1380 for(uint8_t side = 0; side < 2; ++side) {
1381 if(rect_intersect(&bbox, &cur->r[side], &new_isect.isect)) {
1382 isect_found = true;
1383
1384 new_isect.tree_idx = tidx;
1385 new_isect.next = new_lp.side_isect[side];
1386
1387 ALLOC_INFER(new_lp.side_isect[side], 1);
1388 *new_lp.side_isect[side] = new_isect;
1389 ++new_lp.side_count[side];
1390
1391 memset(&new_isect, 0, sizeof new_isect);
1392 }
1393 }
1394 }
1395
1396 // If there is intersection with a tree, add to return.
1397 if(isect_found) {
1398 ++ret->local_portal_count;
1399
1400 new_lp.p = cur;
1401 new_lp.next = ret->start;
1402 new_lp.portal_gid = portal_gid;
1403
1404 ALLOC_INFER(ret->start, 1);
1405 *ret->start = new_lp;
1406
1407 memset(&new_lp, 0, sizeof new_lp);
1408 }
1409 ++portal_gid;
1410 }
1411
1412 ret->by_proc = g_hash_table_new_full(g_direct_hash, g_direct_equal,
1413 NULL, (GDestroyNotify)destroy_portal_tree_isect_list);
1414
1415 return ret;
1416}

1487static void
1488dispatch_search_from_portals(load_balancer *ctx, _FringeBuilder *fb,
1489 TermDetection *td, struct _local_portals *lps, unsigned fringesize)
1490{
1491 int nprocs;
1492 MPI_Comm_size(ctx->group, &nprocs);
1493
1494 int myrank;
1495 MPI_Comm_rank(ctx->group, &myrank);
1496
1497 // Number of portals each process have.
1498 unsigned proc_portal_count[nprocs];
1499 MPI_Allgather(&lps->local_portal_count, 1, MPI_UNSIGNED,
1500 proc_portal_count, 1, MPI_UNSIGNED, ctx->group);
1501
1502 // Portal information sent to other processes
1503 typedef struct {
1504 size_t portal_gid;
1505 uint8_t side0: 1;
1506 uint8_t side1: 1;
1507 } portal_info;
1508
1509 // Global portal information vector
1510 size_t total_portal_instances;
1511 portal_info *pinfos;
1512
1513 // Portal data index, for direct reference
1514 struct _local_portal *pindex[ctx->num_portals];
1515 memset(pindex, 0, sizeof pindex);
1516
1517 // Gather portal info from all processes.
1518 {
1519 // Assemble my local portal information:
1520 portal_info locpinfo[lps->local_portal_count];
1521 struct _local_portal *lp = lps->start;
1522 for(unsigned i = 0; i < lps->local_portal_count; ++i) {
1523 locpinfo[i].portal_gid = lp->portal_gid;
1524 locpinfo[i].side0 = lp->side_isect[0] != NULL;
1525 locpinfo[i].side1 = lp->side_isect[1] != NULL;
1526
1527 assert(!pindex[lp->portal_gid]);
1528 pindex[lp->portal_gid] = lp;
1529
1530 lp = lp->next;
1531 }
1532
1533 // Organize receive buffer parameters
1534 int displs[nprocs];
1535 int counts[nprocs];
1536 total_portal_instances = 0;
1537 for(unsigned i = 0; i < nprocs; ++i) {
1538 displs[i] = total_portal_instances
1539 * sizeof(portal_info);
1540 total_portal_instances += proc_portal_count[i];
1541 counts[i] = proc_portal_count[i] * sizeof(portal_info);
1542 }
1543 ALLOC_INFER(pinfos, total_portal_instances);
1544
1545 MPI_Allgatherv(locpinfo, sizeof locpinfo, MPI_BYTE, pinfos,
1546 counts, displs, MPI_BYTE, ctx->group);
1547 }
1548
1549 allocator *al = allocator_init();
1550 isend_pool *isp = isend_pool_create();
1551
1552 // Look through every received portal_info, and see if there is a match.
1553 unsigned to_recv_count = 0;
1554 unsigned i = 0;
1555 for(int proc = 0; proc < nprocs; ++proc) {
1556 for(unsigned pc = 0; pc < proc_portal_count[proc]; ++pc) {
1557 // Get local instance of given portal:
1558 struct _local_portal *lp = pindex[pinfos[i].portal_gid];
1559 if(lp) {
1560 // Check possible match and send local intersections:
1561 if(lp->side_isect[0] && pinfos[i].side1) {
1562 send_portal_side(ctx, al, isp, lp, 0, proc);
1563 to_recv_count++;
1564 }
1565 if(lp->side_isect[1] && pinfos[i].side0) {
1566 send_portal_side(ctx, al, isp, lp, 1, proc);
1567 to_recv_count++;
1568 }
1569 }
1570
1571 ++i;
1572 }
1573 }
1574
1575 free(pinfos);
1576
1577 // Receive each "to_recv_count" and compare if we have a neighbor
1578 // through a portal.
1579 for(i = 0; i < to_recv_count; ++i) {
1580 MPI_Status s;
1581 MPI_Probe(MPI_ANY_SOURCE, portal_search_tag, ctx->group, &s);
1582
1583 int byte_count;
1584 MPI_Get_count(&s, MPI_BYTE, &byte_count);
1585
1586 struct serialized_portal_isection *rcbuf;
1587 size_t isect_count =
1588 (byte_count - sizeof *rcbuf) / sizeof rcbuf->isects[0];
1589
1590 ALLOC_INFER_FLEX(rcbuf, isects, isect_count);
1591 MPI_Recv(rcbuf, byte_count, MPI_BYTE, s.MPI_SOURCE,
1592 portal_search_tag, ctx->group, MPI_STATUS_IGNORE);
1593
1594 struct _local_portal *lp = pindex[rcbuf->portal_gid];
1595 assert(lp);
1596
1597 uint8_t lside = !rcbuf->side;
1598
1599 // For each intersection between local side and remote side,
1600 // store it, add a remote tree neighbor and dispatch a remote
1601 // fringe search.
1602 struct _portal_tree_isect *it = lp->side_isect[lside];
1603 assert(it);
1604
1605 Point loc2rem;
1606 POINT_SUB(loc2rem,
1607 lp->p->r[rcbuf->side].lo, lp->p->r[lside].lo);
1608
1609 Point rem2loc;
1610 POINT_SUB(rem2loc,
1611 lp->p->r[lside].lo, lp->p->r[rcbuf->side].lo);
1612
1613 struct _search_work_start sworks[lp->side_count[lside] * isect_count];
1614 unsigned sworks_count = 0;
1615
1616 // Get contact list for process, or allocate if doesn't exist.
1617 struct portal_tree_isect_list *proc_contacts =
1618 g_hash_table_lookup(lps->by_proc,
1619 GINT_TO_POINTER(s.MPI_SOURCE));
1620 if(!proc_contacts) {
1621 ALLOC_INFER(proc_contacts, 1);
1622 proc_contacts->start = NULL;
1623 g_hash_table_insert(lps->by_proc,
1624 GINT_TO_POINTER(s.MPI_SOURCE), proc_contacts);
1625 }
1626
1627 while(it) {
1628 Rect disp_lisect;
1629 rect_translate(loc2rem, &it->isect, &disp_lisect);
1630 for(size_t i = 0; i < isect_count; ++i) {
1631 // Skip if groups are different.
1632 if(rcbuf->isects[i].group
1633 != ctx->output[it->tree_idx].group)
1634 {
1635 continue;
1636 }
1637
1638 // For each intersection between local side and
1639 // remote side, store it, add a remote tree
1640 // neighbor and dispatch a remote fringe search.
1641 if(rect_intersect(&disp_lisect, &rcbuf->isects[i].r,
1642 &sworks[sworks_count].isect))
1643 {
1644 struct _search_work_start *sw =
1645 &sworks[sworks_count++];
1646 sw->tree_group = rcbuf->isects[i].group;
1647 sw->search_from_dist = fringesize;
1648 POINT_ASSIGN(sw->disp, loc2rem);
1649
1650 DECL_AND_ALLOC(struct _portal_tree_isect, new, 1);
1651 new->next = proc_contacts->start;
1652 proc_contacts->start = new;
1653 new->tree_idx = it->tree_idx;
1654
1655 rect_translate(rem2loc, &sw->isect,
1656 &new->isect);
1657
1658 _fringe_builder_add_remote_tree_neighbor(fb,
1659 s.MPI_SOURCE, it->tree_idx, loc2rem,
1660 &new->isect);
1661 }
1662 }
1663
1664 it = it->next;
1665 }
1666
1667 if(sworks_count) {
1668 _dispatch_remote_fringe_search(td, s.MPI_SOURCE,
1669 myrank, sworks_count, sworks);
1670 }
1671
1672 free(rcbuf);
1673 }
1674
1675 isend_pool_destroy(isp);
1676 allocator_destroy(al);
1677}

1679static void
1680_parallel_extend_fringe(load_balancer *ctx, partition_graph *pg)
1681{
1682 // Find which half-portals belong to this process:
1683 struct _local_portals *lps = _local_portals_create(ctx);
1684
1685 // Find the process bounding box intersection
1686 struct _proc_conectivity *pc = _proc_conn_parallel_find(ctx);
1687
1688 DECL_AND_ALLOC(struct _search_work_start, isects, ctx->num_out_higs * pc->num_neighbors);
1689 DECL_AND_ALLOC(unsigned, touching_trees, ctx->num_out_higs * pc->num_neighbors);
1690 DECL_AND_ALLOC(unsigned, isects_per_proc_count, pc->num_neighbors);
1691 DECL_AND_ALLOC(unsigned, proc_start, pc->num_neighbors);
1692 memset(isects_per_proc_count, 0, pc->num_neighbors * sizeof *isects_per_proc_count);
1693
1694 TermDetection *td = term_det_init(ctx->group);
1695
1696 // Creates the structure holding the in construction fringes for all relevant
1697 // neighbouring processes.
1698 _FringeBuilder *fb = _fringe_builder_create(pc, lps, ctx, td);
1699
1700 // Find where each tree touches other processes.
1701 {
1702 int myrank;
1703 MPI_Comm_rank(ctx->group, &myrank);
1704 Point zero_disp = {};
1705
1706 unsigned next_isect = 0;
1707 for(size_t j = 0; j < pc->num_neighbors; ++j) {
1708 proc_start[j] = next_isect;
1709
1710 for(size_t i = 0; i < ctx->num_out_higs; ++i) {
1711 // If doesn't have fringe, skip it
1712 if(!ctx->group_fringe[ctx->output[i].group])
1713 continue;
1714
1715 Rect tree_box;
1716 hig_get_bounding_box(ctx->output[i].tree, &tree_box);
1717
1718 if(rect_intersect(&pc->neighbors[j].intersection, &tree_box,
1719 &isects[next_isect].isect))
1720 {
1721 isects[next_isect].search_from_dist = pg->fringesize;
1722 isects[next_isect].tree_group = ctx->output[i].group;
1723 POINT_ASSIGN_SCALAR(isects[next_isect].disp, 0);
1724
1725 touching_trees[next_isect] = i;
1726 const int nb_rank = pc->neighbors[j].proc;
1727 _fringe_builder_add_remote_tree_neighbor(fb, nb_rank,
1728 i, zero_disp, &isects[next_isect].isect);
1729
1730 // Handle the special case a intersecting flat tree is
1731 // coplanar with a process boundary, in which case
1732 // a fringe copy of the local tree must be available
1733 // in the remote process. This is needed to properly
1734 // handle boundary conditions that are separated from
1735 // their domain by the partition.
1736 if(rect_are_parallel(&tree_box,
1737 &pc->neighbors[j].intersection))
1738 {
1739 Point disp = {0};
1740 _fringe_builder_search_from(fb,
1741 &isects[next_isect].isect, 1,
1742 pc->neighbors[j].proc, disp, i);
1743 }
1744
1745 ++next_isect;
1746 ++isects_per_proc_count[j];
1747 }
1748 }
1749
1750 // Send the found intersections to the other processes
1751 if(isects_per_proc_count[j]) {
1752 _dispatch_remote_fringe_search(td, pc->neighbors[j].proc,
1753 myrank, isects_per_proc_count[j],
1754 &isects[proc_start[j]]);
1755 }
1756 }
1757 }
1758
1759 // Jumpstart seach from portals, both inside and outside current
1760 // process. Don't forget to _fringe_builder_add_remote_tree_neighbor for
1761 // portal neighbors.
1762 dispatch_search_from_portals(ctx, fb, td, lps, pg->fringesize);
1763
1764 // Receive tree intersections from neighbors, compute their
1765 // fringes and send the result back.
1766 {
1767 int from;
1768 int msg_size;
1769 struct _search_work *work;
1770 while((work = term_det_wait_for_msg(td, &from, &msg_size, true))) {
1771 size_t count = (msg_size - sizeof *work) / sizeof work->sstarts[0];
1772
1773 // Get proc idx, if geometrically connected:
1774 unsigned geom_proc;
1775 bool geom_conn = _fringe_builder_get_neighbor_idx(fb, from, &geom_proc);
1776
1777 // Get per process portal connection info:
1778 struct _portal_tree_isect *ppc =
1779 _local_portals_get_by_proc(lps, from);
1780
1781 // We already have the local intersections with that process,
1782 // which are the same rects sent to it. Check each new received
1783 // rect with the rects sent to that process.
1784 for(size_t i = 0; i < count; ++i) {
1785 struct _search_work_start *ss = &work->sstarts[i];
1786
1787 // Check the geometrical connections:
1788 if(geom_conn) {
1789 for(size_t k = 0; k < isects_per_proc_count[geom_proc];
1790 ++k)
1791 {
1792 size_t idx = k + proc_start[geom_proc];
1793
1794 Rect isect;
1795 const unsigned tid = touching_trees[idx];
1796 if(ss->tree_group != ctx->output[tid].group
1797 || !rect_intersect(&ss->isect,
1798 &isects[idx].isect, &isect))
1799 {
1800 continue;
1801 }
1802
1803 // The areas received and sent touch each
1804 // other at isect, and the trees are of the
1805 // same group.
1806 // Build the remote fringe from the local
1807 // tree, and span the search to other
1808 // neighboring (possibly remote) trees.
1809 _fringe_builder_search_from(
1810 fb, &isect,
1811 work->sstarts[i].search_from_dist,
1812 work->answer_to_proc,
1813 work->sstarts[i].disp,
1814 tid
1815 );
1816 }
1817 }
1818
1819 // Check portal connections:
1820 struct _portal_tree_isect *it =
1821 _local_portals_get_by_proc(lps, from);
1822 for(; it; it = it->next) {
1823 if(ctx->output[it->tree_idx].group != ss->tree_group) {
1824 continue;
1825 }
1826
1827 Rect isect;
1828 if(!rect_intersect(&ss->isect, &it->isect, &isect)) {
1829 continue;
1830 }
1831
1832 _fringe_builder_search_from(fb, &isect,
1833 work->sstarts[i].search_from_dist,
1834 work->answer_to_proc,
1835 work->sstarts[i].disp,
1836 it->tree_idx
1837 );
1838 }
1839 }
1840 }
1841 }
1842 term_det_destroy(td);
1843
1844 free(isects_per_proc_count);
1845 free(proc_start);
1846 free(isects);
1847 free(touching_trees);
1848
1849 // By now, this process have in its FringeBuilder the fringe it must provide to
1850 // all remote processes (not just neighbors' fringes, but any process' that might
1851 // span as far as here).
1852
1853 // Build the tree slices of the fringe's cells and sync them with the neighboring
1854 // processed.
1855 {
1856 unsigned count;
1857 struct _lb_output_tree *fringes = _fringe_builder_assemble_result(fb, &count, pg);
1858
1859 // Add the fringe trees to the output.
1860 REALLOC_INFER(ctx->output, ctx->num_out_higs + count);
1861 memcpy(ctx->output + ctx->num_out_higs, fringes, count * sizeof *fringes);
1862 ctx->num_out_higs += count;
1863 }
1864
1865 _fringe_builder_destroy(fb);
1866 _proc_conn_destroy(pc);
1867 _local_portals_destroy(lps);
1868}

1870void lb_calc_partition(load_balancer *ctx, partition_graph *pg)
1871{
1872 // First, distribute the trees
1873 _parallel_partition_and_distribute_trees(ctx);
1874 assert(ctx->trees == NULL);
1875
1876 // Index the base trees properties in the partition_graph
1877 for(unsigned i = 0; i < ctx->num_out_higs; ++i)
1878 {
1879 DECL_AND_ALLOC(struct tree_properties, prop, 1);
1880 prop->is_fringe = false;
1881 prop->partition_group = ctx->output[i].group;
1882
1883 g_hash_table_insert(pg->tree_props, ctx->output[i].tree, prop);
1884 }
1885
1886 // Then extend the output to include fringe sizes
1887 if(pg_get_fringe_size(pg) > 0)
1888 _parallel_extend_fringe(ctx, pg);
1889
1890 // Sort output trees according to its group
1891 qsort(ctx->output, ctx->num_out_higs, sizeof *ctx->output, (int(*)(const void *, const void *))_out_group_cmp);
1892
1893 // Count how many trees are there in each group, and calculate the
1894 // stating index for the groups:
1895 assert(ctx->out_higs_group_start == NULL);
1896 ALLOC_INFER(ctx->out_higs_group_start, ctx->num_groups + 1);
1897 memset(ctx->out_higs_group_start, 0,
1898 (ctx->num_groups + 1) * sizeof *ctx->out_higs_group_start);
1899
1900 // First just count how many trees in each group
1901 for(size_t i = 0; i < ctx->num_out_higs; ++i) {
1902 ctx->out_higs_group_start[ctx->output[i].group + 1]++;
1903 }
1904
1905 // Then calculate the starting index.
1906 for(size_t i = 1; i <= ctx->num_groups; ++i) {
1907 ctx->out_higs_group_start[i] += ctx->out_higs_group_start[i-1];
1908 }
1909}
