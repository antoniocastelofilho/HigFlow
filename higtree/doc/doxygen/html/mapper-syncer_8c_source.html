100 // First, the asynchronous send loop,
101 GHashTableIter iter;
102 g_hash_table_iter_init (&iter, psd->filtered_neighbors);
103
104 gpointer key;
105 struct neighbor_proc *np;
106 while (g_hash_table_iter_next (&iter, &key, (gpointer *)&np)) {
107 int proc = GPOINTER_TO_INT(key);
108 // Organize and send data to neighbor process.
109 unsigned count;
110 send_buff[send_count] = _ms_organize_send(ms, sd, np, &count);
111 MPI_Isend(send_buff[send_count], count, MPI_INT, proc,
112 _ms_tag, comm, &send_reqs[send_count]);
113 ++send_count;
114
115 // Before going on, we try to receive as much pending messages
116 // as we can, so to not leave the network buffers full.
117 // TODO: test if this makes any difference in runtime or not
118 for(;;) {
119 MPI_Status status;
120 int recv_ready = 0;
121 MPI_Iprobe(MPI_ANY_SOURCE, _ms_tag, comm, &recv_ready, &status);
122 if(recv_ready) {
123 _ms_receive(ms, sd, &status);
124 ++recv_count;
125 } else {
126 break;
127 }
128 }
129 }
130
131 // Then receive what hasn't been received yet
132 for(; recv_count < nbs_count; ++recv_count) {
133 MPI_Status status;
134 MPI_Probe(MPI_ANY_SOURCE, _ms_tag, comm, &status);
135 _ms_receive(ms, sd, &status);
136 }
137
138 // Sending cleanup
139 assert(send_count == nbs_count);
140 MPI_Waitall(send_count, send_reqs, MPI_STATUSES_IGNORE);
