16void mp_set_numbits(mp_mapper *m, int numbits) {
17 assert(numbits <= MP_32_BITS);
18 assert(
19 numbits == MP_1_BIT ||
20 numbits == MP_2_BITS ||
21 numbits == MP_4_BITS ||
22 numbits == MP_8_BITS ||
23 numbits == MP_16_BITS ||
24 numbits == MP_32_BITS
25 );
26 m->numbits = numbits;
27 m->numvals_per_pos = sizeof(GPOINTER_TO_INT(NULL))*8/numbits;

34void mp_assign(mp_mapper *m, uniqueid id, mp_value_t val) {
35 if (m->numbits == MP_ALL_BITS) {
36 g_hash_table_insert(m->map, GINT_TO_POINTER(id), GINT_TO_POINTER(val+1));
37 } else {
38 DEBUG_PASS;
39 assert(val < (1 << m->numbits));
40 uniqueid new_id = id / m->numvals_per_pos;
41 mp_value_t v = GPOINTER_TO_INT(g_hash_table_lookup(m->map, GINT_TO_POINTER(new_id)));
42 int pos_id = id % m->numvals_per_pos;
43 int shift = m->numbits * pos_id;
44 mp_value_t new_v = (v & ~(m->mask << shift)) | (val << shift);
45 g_hash_table_insert(m->map, GINT_TO_POINTER(new_id), GINT_TO_POINTER(new_v));
46 }
47 if (id > m->largest_key) {
48 m->largest_key = id;
49 }
50}

52mp_value_t mp_lookup(mp_mapper *m, uniqueid id) {
53 if (m->numbits == MP_ALL_BITS) {
54 gpointer *col = g_hash_table_lookup(m->map, GINT_TO_POINTER(id));
55 return (mp_value_t) (GPOINTER_TO_INT(col) - 1);
56 } else {
57 uniqueid new_id = id / m->numvals_per_pos;
58 mp_value_t v = GPOINTER_TO_INT(g_hash_table_lookup(m->map, GINT_TO_POINTER(new_id)));
59 int pos_id = id % m->numvals_per_pos;
60 int shift = m->numbits * pos_id;
61 mp_value_t new_v = (v >> shift) % (1 << m->numbits);
62 return new_v;
63 }
