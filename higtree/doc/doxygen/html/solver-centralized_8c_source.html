31{
32 _solver_centralized *s = scast(cs);

124static void master_assemble_matrix(struct solver *cs)
125{
126 _solver_centralized *s = scast(cs);
127 _solver_centralized_split *psplits = s->master->all;
128
129 int num_procs;
130 MPI_Comm_size(MPI_COMM_WORLD, &num_procs);
131
132 unsigned int *row_jumper;
133 unsigned int *col_buffer;
134 real *coefs;
135
136 row_jumper = malloc((s->master->total_size+1) * sizeof(unsigned int));
137 row_jumper[0] = 0;
138
139 // Get the start point for remote rows
140 MPI_Request *reqs = malloc(2 * (num_procs-1) * sizeof(MPI_Request));
141 for(int i = 1; i < num_procs; ++i) {
142 _solver_centralized_split *psplit = &psplits[i-1];
143 MPI_Irecv(&row_jumper[psplit->start+1], psplit->size, MPI_UNSIGNED, i, INT_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[i-1]);
144 }
145
146 // Calculate start point for local rows
147 local_row_size_assemble(s, &row_jumper[s->local.start+1]);
148
149 MPI_Waitall(num_procs-1, reqs, MPI_STATUSES_IGNORE);
150
151 _solver_centralized_split proc_elems[num_procs];
152
153 // Normalize starting points with global index
154 size_t total_elems;
155 {
156 unsigned offset = 0;
157 unsigned count = 0;
158 const unsigned int num_jumpers = s->master->total_size + 1;
159 int* procs_order = s->master->procs_order;
160
161 proc_elems[procs_order[0]].start = 0;
162 for(unsigned int i = 1; i < num_jumpers; ++i) {
163 row_jumper[i] += offset;
164 if(row_jumper[i] < row_jumper[i-1]) {
165 unsigned int size = row_jumper[i-1] - offset;
166 row_jumper[i] += size;
167 proc_elems[procs_order[count]].size = size;
168 offset = row_jumper[i-1];
169 proc_elems[procs_order[++count]].start = offset;
170 }
171
172 }
173 total_elems = row_jumper[num_jumpers-1];
174 proc_elems[procs_order[count]].size = total_elems - offset;
175 assert(count == num_procs-1);
176 }
177
178 // Receive the bulk data (coefficients and colums)
179 col_buffer = malloc(total_elems * sizeof(unsigned int));
180 coefs = malloc(total_elems * sizeof(real));
181 for(int i = 1; i < num_procs; ++i) {
182 size_t reqi = 2*(i-1);
183 _solver_centralized_split *elems = &proc_elems[i];
184 MPI_Irecv(&col_buffer[elems->start], elems->size, MPI_UNSIGNED, i, INT_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[reqi]);
185 MPI_Irecv(&coefs[elems->start], elems->size, get_mpi_real(), i, REAL_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[reqi+1]);
186
187 }
188
189 // Assemble the local bulk data
190 local_row_assemble(s, &col_buffer[proc_elems[0].start], &coefs[proc_elems[0].start]);
191
192 MPI_Waitall(2 * (num_procs-1), reqs, MPI_STATUSES_IGNORE);
193 free(reqs);
194
195 s->cvtable->set_full_A(s, row_jumper, col_buffer, coefs, total_elems);
196
197 free(coefs);
198 free(col_buffer);
199 free(row_jumper);
200}

202static void master_vector_assemble(_solver_centralized *s, real *vec, real *local_buf)
203{
204 int num_procs;
205 MPI_Comm_size(MPI_COMM_WORLD, &num_procs);
206
207 // Get remote
208 MPI_Request reqs[num_procs-1];
209 for(int i = 1; i < num_procs; ++i) {
210 _solver_centralized_split *psplit = &s->master->all[i-1];
211 MPI_Irecv(&vec[psplit->start], psplit->size, get_mpi_real(), i, REAL_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[i-1]);
212 }
213
214 // Copy local
215 memcpy(&vec[s->local.start], local_buf, s->local.size * sizeof(real));
216
217 // Wait remote copy to finish
218 MPI_Waitall(num_procs-1, reqs, MPI_STATUSES_IGNORE);
219}

240static void master_solve(struct solver *cs)
241{
242 _solver_centralized *s = scast(cs);
243 real *sol = s->master->solution;
244
245 // Solver the previously configured system
246 s->cvtable->solve(s, sol);
247
248 // Send the result back to the processes
249 int num_procs;
250 MPI_Comm_size(MPI_COMM_WORLD, &num_procs);
251
252 MPI_Request reqs[num_procs];
253 for(int i = 1; i < num_procs; ++i) {
254 _solver_centralized_split *psplit = &s->master->all[i-1];
255 MPI_Isend(&sol[psplit->start], psplit->size, get_mpi_real(), i, REAL_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[i-1]);
256 }
257
258 // Copy local
259 memcpy(s->X_cpu, &sol[s->local.start], s->local.size * sizeof(real));
260
261 // Wait remote copy to finish
262 MPI_Waitall(num_procs-1, reqs, MPI_STATUSES_IGNORE);
263}

270static void slave_assemble_matrix(struct solver *cs)
271{
272 _solver_centralized *s = scast(cs);
273 unsigned int *row_jumper = malloc(s->local.size * sizeof(unsigned int));
274
275 local_row_size_assemble(s, row_jumper);
276 MPI_Request req[3];
277 MPI_Isend(row_jumper, s->local.size, MPI_UNSIGNED, 0, INT_EXCHANGE_TAG, MPI_COMM_WORLD, &req[0]);
278
279 size_t elem_count = row_jumper[s->local.size-1];
280 unsigned int *col_buffer = malloc(elem_count * sizeof(unsigned int));
281 real *coefs = malloc(elem_count * sizeof(real));
282 local_row_assemble(s, col_buffer, coefs);
283
284 MPI_Isend(col_buffer, elem_count, MPI_UNSIGNED, 0, INT_EXCHANGE_TAG, MPI_COMM_WORLD, &req[1]);
285 MPI_Isend(coefs, elem_count, get_mpi_real(), 0, REAL_EXCHANGE_TAG, MPI_COMM_WORLD, &req[2]);
286
287 MPI_Waitall(3, req, MPI_STATUSES_IGNORE);
288
289 free(coefs);
290 free(col_buffer);
291 free(row_jumper);
292}

403void _solver_centralized_init_master(_solver_centralized *slv, size_t start, size_t size, const _solver_centralized_vtable *internal_vtable) {
404 _solver_centralized_init(slv, start, size, internal_vtable);
405
406 slv->s.vtable = &master_vtable;
407
408 // Uses MPI to get to know other processes split:
409
410 int num_procs;
411 MPI_Comm_size(MPI_COMM_WORLD, &num_procs);
412 _solver_centralized_master *master_data;
413 master_data = slv->master = malloc(sizeof(_solver_centralized_master)
414 + (num_procs-1) * sizeof(_solver_centralized_split));
415
416 master_data->procs_order = malloc(num_procs * sizeof(int));
417
418 MPI_Request *reqs = malloc((num_procs-1) * sizeof(MPI_Request));
419 for(int i = 1; i < num_procs; ++i) {
420 MPI_Irecv(master_data->all[i-1].serial, 2, MPI_UNSIGNED,
421 i, INT_EXCHANGE_TAG, MPI_COMM_WORLD, &reqs[i-1]);
422 }
423 MPI_Waitall(num_procs-1, reqs, MPI_STATUSES_IGNORE);
424 free(reqs);
425
426 unsigned long total_size = size;
427 ProcOrderer order[num_procs];
428 for(int i = 1; i < num_procs; ++i) {
429 total_size += master_data->all[i-1].size;
430 order[i].start = master_data->all[i-1].start;
431 order[i].proc_id = i;
432 }
433 master_data->total_size = total_size;
434 master_data->solution = malloc(total_size * sizeof(real));
435
436 order[0].start = slv->local.start;
437 order[0].proc_id = 0;
438 qsort(order, num_procs, sizeof(ProcOrderer),
439 (int (*)(const void *, const void *))proc_compare);
440
441 for(int i = 0; i < num_procs; ++i)
442 {
443 master_data->procs_order[i] = order[i].proc_id;
444 }
445}
