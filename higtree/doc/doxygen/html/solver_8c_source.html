54void slv_set_Aij(solver *s, int i, int j, real v) {
55
56 // Preallocate before setting values
57 if (s->MaxNonZeros == 0) {
58 slv_set_maxnonzeros(s, 1000);
59 }
60
61 if (i == s->lasti) {
62
63 // Cache values to be added at once in the matrix
64 s->j [s->lastj] = j;
65 s->vs[s->lastj] = v;
66 s->lastj++;
67 }
68
69 else {
70
71 if (s->lasti != -1) {
72 // Refrain from setting an imposed value
73 if(s->imposed_line != s->lasti) {
74 // Add cached values to the matrix
75 s->vtable->set_Ai(s, s->lasti, s->lastj, s->j, s->vs);
76 }
77
78 // Reset pointers
79 for (int c = 0; c < s->lastj; c++) {
80 s->j[c] = 0;
81 s->vs[c] = 0.0;
82 }
83 } else {
84 // The cache is not allocated
85 // allocate it!
86 ALLOC(int, s->j, s->size);
87 ALLOC(real, s->vs, s->size);
88 }
89
90 // Now cache new value
91 s->lasti = i;
92 s->j [0] = j;
93 s->vs[0] = v;
94 s->lastj = 1;
95 }
